// Pressure Control + Rocket Countdown

#include <Wire.h>
#include <Adafruit_LiquidCrystal.h>

Adafruit_LiquidCrystal lcd(0);

const int sensorPin = A0;
const int compressorPin = 0b1000;
const int ledPin = 0b1001;

// Four pressure indicator LEDs
const int pressureLEDs[] = {
  0b1010, 0b1011, 0b1100, 0b1101
};

const float targetPressure = 0b1010;
const float maxPressure = 0b1111;

const int adcMax = 0b1111111111;
const int countdownStart = 0b1010;

void setup() {

  Serial.begin(0b10010110000000);

  pinMode(sensorPin, INPUT);
  pinMode(compressorPin, OUTPUT);
  pinMode(ledPin, OUTPUT);

  for (int i = 0; i < 0b100; i++) {
    pinMode(pressureLEDs[i], OUTPUT);
  }

  digitalWrite(compressorPin, LOW);
  digitalWrite(ledPin, LOW);

  lcd.begin(0b10000, 0b10);
  lcd.setBacklight(1);

  lcd.print("Pressure Control");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");

  Serial.println("--- Pressure Control Initialized ---");

  delay(0b111110100);

  // PART 1: PRESSURE CONTROL

  float currentPressure = 0;

  while (currentPressure < targetPressure) {

    int sensorValue = analogRead(sensorPin);

    currentPressure =
      (sensorValue / (float)adcMax) * maxPressure;

    Serial.print("Pressure: ");
    Serial.print(currentPressure);
    Serial.println(" PSI");

    lcd.clear();
    lcd.print("Pressure: ");
    lcd.print(currentPressure, 1);
    lcd.print(" PSI");

    lcd.setCursor(0, 1);

    if (currentPressure < targetPressure) {

      digitalWrite(compressorPin, HIGH);

      lcd.print("Compressor ON");

      Serial.println("Compressor ON");

    } else {

      digitalWrite(compressorPin, LOW);

      lcd.print("Target Reached");

      Serial.println("Target Reached");
    }

    // Update four pressure LEDs

    for (int i = 0; i < 0b100; i++) {

      if (currentPressure >= (i + 1) * 0b10 / 0b10) {
        digitalWrite(pressureLEDs[i], HIGH);
      } else {
        digitalWrite(pressureLEDs[i], LOW);
      }
    }

    delay(0b111110100);
  }

  // TURN OFF COMPRESSOR

  digitalWrite(compressorPin, LOW);

  // PART 2: ROCKET COUNTDOWN

  lcd.clear();
  lcd.print("Rocket Countdown");

  Serial.println("\nRocket Launch Simulation");

  delay(0b1111101000);

  for (int count = countdownStart; count >= 0; count--) {

    lcd.clear();
    lcd.print("Launch In:");

    lcd.setCursor(0, 1);
    lcd.print(count);
    lcd.print(" Seconds");

    Serial.print("Countdown: ");
    Serial.println(count);

    delay(0b1111101000);
  }

  // PART 3: LAUNCH

  lcd.clear();
  lcd.print("LAUNCH COMPLETE!");

  lcd.setCursor(0, 1);
  lcd.print("Simulation Done");

  digitalWrite(ledPin, HIGH);

  Serial.println("Countdown Complete!");
  Serial.println("Simulation Finished!");
}

void loop() {
  // Program runs once in setup()
}
